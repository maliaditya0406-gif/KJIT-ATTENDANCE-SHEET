"use strict"; // ========================================== // KJIT CAMPUS - ATTENDANCE REPORTS // ========================================== const STUDENTS_KEY = "kjitStudents"; const ATTENDANCE_KEY = "attendanceData"; const DARK_MODE_KEY = "kjitDarkMode"; // ========================================== // LOAD DATA // ========================================== function getStudents() { try { const data = localStorage.getItem(STUDENTS_KEY); if (!data) { return []; } const students = JSON.parse(data); return Array.isArray(students) ? students : []; } catch (error) { console.error("Error loading students:", error); return []; } } function getAttendanceData() { try { const data = localStorage.getItem(ATTENDANCE_KEY); if (!data) { return {}; } const attendance = JSON.parse(data); return attendance && typeof attendance === "object" ? attendance : {}; } catch (error) { console.error("Error loading attendance data:", error); return {}; } } // ========================================== // GLOBAL DATA // ========================================== let students = getStudents(); let attendanceData = getAttendanceData(); // ========================================== // ELEMENTS // ========================================== const reportDate = document.getElementById("reportDate"); const reportSearch = document.getElementById("reportSearch"); const reportBody = document.getElementById("reportBody"); const exportReportButton = document.getElementById("exportReportButton"); const printReportButton = document.getElementById("printReportButton"); const themeButton = document.getElementById("themeButton"); // ========================================== // TODAY'S DATE // ========================================== function setToday() { if (!reportDate) { return; } const today = new Date(); const year = today.getFullYear(); const month = String( today.getMonth() + 1 ).padStart(2, "0"); const day = String( today.getDate() ).padStart(2, "0"); reportDate.value = `${year}-${month}-${day}`; } // ========================================== // CALCULATE STUDENT ATTENDANCE // ========================================== function calculateAttendance(roll, date) { let present = 0; let absent = 0; if (!date) { return { present: 0, absent: 0, total: 0, percentage: 0 }; } /* Expected attendance format: attendanceData = { "2026-08-22": { "1": { "1": "P", "2": "A", "3": "P", "4": "P", "5": "A", "6": "P" } } } */ const dateData = attendanceData[date]; if (!dateData) { return { present: 0, absent: 0, total: 0, percentage: 0 }; } // Convert roll number to string // because localStorage object keys are strings. const studentData = dateData[String(roll)] || dateData[roll]; if (!studentData) { return { present: 0, absent: 0, total: 0, percentage: 0 }; } // Six periods for (let period = 1; period <= 6; period++) { const status = studentData[period] ?? studentData[String(period)]; if (status === "P") { present++; } else if (status === "A") { absent++; } } const total = present + absent; const percentage = total > 0 ? Math.round((present / total) * 100) : 0; return { present, absent, total, percentage }; } // ========================================== // STATUS // ========================================== function getStatus(percentage) { if (percentage >= 75) { return { text: "Good", className: "good" }; } if (percentage >= 50) { return { text: "Warning", className: "warning" }; } return { text: "Low", className: "danger" }; } // ========================================== // RENDER REPORT // ========================================== function renderReport() { if (!reportBody) { console.warn( "reportBody element was not found." ); return; } // Reload latest data students = getStudents(); attendanceData = getAttendanceData(); // Clear old rows reportBody.innerHTML = ""; const date = reportDate ? reportDate.value : ""; const search = reportSearch ? reportSearch.value .toLowerCase() .trim() : ""; // Active students only const activeStudents = students.filter( student => student && student.active !== false ); let totalPresent = 0; let totalAbsent = 0; // ====================================== // NO STUDENTS // ====================================== if (activeStudents.length === 0) { const row = document.createElement("tr"); const cell = document.createElement("td"); cell.colSpan = 7; cell.textContent = "No students found."; cell.style.textAlign = "center"; row.appendChild(cell); reportBody.appendChild(row); updateSummary(0, 0); return; } // ====================================== // STUDENTS // ====================================== activeStudents.forEach(student => { if (!student) { return; } const studentName = String( student.name ?? "" ); const studentRoll = String( student.roll ?? "" ); // Search filter if ( search && !studentName .toLowerCase() .includes(search) && !studentRoll .includes(search) ) { return; } const stats = calculateAttendance( student.roll, date ); totalPresent += stats.present; totalAbsent += stats.absent; // ================================== // CREATE ROW // ================================== const row = document.createElement("tr"); // ================================== // ROLL NUMBER // ================================== const rollCell = document.createElement("td"); rollCell.textContent = studentRoll ? studentRoll.padStart(2, "0") : "-"; // ================================== // STUDENT NAME // ================================== const nameCell = document.createElement("td"); nameCell.textContent = studentName || "Unknown Student"; nameCell.className = "student-name"; // ================================== // PRESENT // ================================== const presentCell = document.createElement("td"); presentCell.textContent = stats.present; // ================================== // ABSENT // ================================== const absentCell = document.createElement("td"); absentCell.textContent = stats.absent; // ================================== // TOTAL // ================================== const totalCell = document.createElement("td"); totalCell.textContent = stats.total; // ================================== // PERCENTAGE // ================================== const percentageCell = document.createElement("td"); percentageCell.textContent = `${stats.percentage}%`; if (stats.percentage >= 75) { percentageCell.className = "percentage-good"; } else if (stats.percentage >= 50) { percentageCell.className = "percentage-warning"; } else { percentageCell.className = "percentage-danger"; } // ================================== // STATUS // ================================== const statusCell = document.createElement("td"); const status = getStatus( stats.percentage ); const badge = document.createElement("span"); badge.className = `status ${status.className}`; badge.textContent = status.text; statusCell.appendChild( badge ); // ================================== // ADD CELLS // ================================== row.appendChild(rollCell); row.appendChild(nameCell); row.appendChild(presentCell); row.appendChild(absentCell); row.appendChild(totalCell); row.appendChild(percentageCell); row.appendChild(statusCell); // Add row reportBody.appendChild(row); }); // ====================================== // NO SEARCH RESULTS // ====================================== if ( reportBody.children.length === 0 ) { const row = document.createElement("tr"); const cell = document.createElement("td"); cell.colSpan = 7; cell.textContent = "No matching students found."; cell.style.textAlign = "center"; row.appendChild(cell); reportBody.appendChild(row); } // ====================================== // UPDATE SUMMARY // ====================================== updateSummary( totalPresent, totalAbsent ); } // ========================================== // UPDATE SUMMARY // ========================================== function updateSummary( present, absent ) { const total = present + absent; const percentage = total > 0 ? Math.round( (present / total) * 100 ) : 0; const activeStudents = students.filter( student => student && student.active !== false ); setText( "reportStudents", activeStudents.length ); setText( "reportPresent", present ); setText( "reportAbsent", absent ); setText( "reportPercentage", `${percentage}%` ); } // ========================================== // HELPER - SET TEXT // ========================================== function setText(id, value) { const element = document.getElementById(id); if (element) { element.textContent = value; } } // ========================================== // SEARCH // ========================================== if (reportSearch) { reportSearch.addEventListener( "input", renderReport ); } // ========================================== // DATE CHANGE // ========================================== if (reportDate) { reportDate.addEventListener( "change", renderReport ); } // ========================================== // EXPORT CSV REPORT // ========================================== function exportReport() { const date = reportDate ? reportDate.value : ""; if (!date) { alert( "Please select a date first." ); return; } students = getStudents(); attendanceData = getAttendanceData(); const activeStudents = students.filter( student => student && student.active !== false ); if (activeStudents.length === 0) { alert( "No students available to export." ); return; } const rows = []; // CSV header rows.push([ "Roll No", "Student Name", "Present", "Absent", "Total", "Percentage", "Status" ]); // Student data activeStudents.forEach(student => { const stats = calculateAttendance( student.roll, date ); const status = getStatus( stats.percentage ); rows.push([ student.roll, student.name, stats.present, stats.absent, stats.total, `${stats.percentage}%`, status.text ]); }); // ====================================== // CREATE CSV // ====================================== const csv = rows .map(row => row .map(value => { const safeValue = String( value ?? "" ) .replace( /"/g, '""' ); return `"${safeValue}"`; }) .join(",") ) .join("\n"); // ====================================== // DOWNLOAD // ====================================== const blob = new Blob( [csv], { type: "text/csv;charset=utf-8;" } ); const url = URL.createObjectURL(blob); const link = document.createElement("a"); link.href = url; link.download = `KJIT-Attendance-Report-${date}.csv`; document.body.appendChild(link); link.click(); document.body.removeChild(link); setTimeout(() => { URL.revokeObjectURL(url); }, 100); } // ========================================== // EXPORT BUTTON // ========================================== if (exportReportButton) { exportReportButton.addEventListener( "click", exportReport ); } // ========================================== // PRINT REPORT // ========================================== if (printReportButton) { printReportButton.addEventListener( "click", () => { window.print(); } ); } // ========================================== // DARK MODE // ========================================== function applySavedTheme() { const darkMode = localStorage.getItem( DARK_MODE_KEY ); if (darkMode === "true") { document.body.classList.add( "dark" ); } else { document.body.classList.remove( "dark" ); } } function toggleReportTheme() { document.body.classList.toggle( "dark" ); const enabled = document.body.classList.contains( "dark" ); localStorage.setItem( DARK_MODE_KEY, enabled ? "true" : "false" ); } if (themeButton) { themeButton.addEventListener( "click", toggleReportTheme ); } // ========================================== // INITIALIZE // ========================================== applySavedTheme(); setToday(); renderReport(); // ========================================== // DEBUG INFORMATION // ========================================== console.log( "KJIT Report System Loaded" ); console.log( "Students:", students ); console.log( "Attendance Data:", attendanceData );
